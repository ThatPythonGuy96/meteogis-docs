def clip(input_file, mask_path):
    ext = os.path.splitext(input_file)[1].lower()
    
    mask_gdf = gpd.read_file(mask_path)

    if ext in (".geojson", ".gpkg", ".shp", ".zip"):
        gdf = gpd.read_file(input_file)

        if gdf.crs is not None and gdf.crs != mask_gdf.crs:
            mask_gdf = mask_gdf.to_crs(gdf.crs)

        clipped = gpd.clip(gdf, mask_gdf)

        # .shp/.zip inputs write out as .geojson — a clipped shapefile
        # needs its own sidecar bundle, which is more complexity than
        # a single-file output format is worth here.
    else:
        raise ValueError(f"Unsupported file type: {ext}")

    return clipped
