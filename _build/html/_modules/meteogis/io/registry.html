def register(driver_cls):
    """
    Class decorator — registers a Driver subclass by its declared extensions.

    Usage:
        @register
        class GeoJSONDriver(Driver):
            extensions = [".geojson", ".json"]
            ...
    """
    for ext in driver_cls.extensions:
        _DRIVERS[ext.lower()] = driver_cls
    return driver_cls

def get_driver(path):
    """
    Look up the Driver class for a file path, based on its extension.

    Parameters
    ----------
    path : str

    Returns
    -------
    Driver
        An instance of the matching driver.

    Raises
    ------
    ValueError
        If no driver is registered for the file's extension.
    """
    import os
    ext = os.path.splitext(path)[1].lower()

    driver_cls = _DRIVERS.get(ext)
    if driver_cls is None:
        raise ValueError(
            f"No driver registered for extension '{ext}'. "
            f"Supported: {sorted(_DRIVERS.keys())}"
        )

    return driver_cls()




# Importing drivers here — after register()/get_driver() are already
# defined above — guarantees the registry is always populated no matter
# which module (raster.py, vector.py, or registry.py itself) gets
# imported first. Every driver's @register decorator runs as a side
# effect of this single line.
from meteogis.io import drivers  # noqa: E402, F401
