"""
Raster I/O built on rasterio.

rasterio wraps GDAL internally but ships portable, self-contained
wheels for Windows/Linux/macOS — so this module needs no system
GDAL install on the end user's machine, unlike the old osgeo-based
version.
"""

from dataclasses import dataclass
from typing import Optional

import numpy as np
import rasterio
from rasterio.errors import RasterioIOError
from rasterio.transform import Affine

@dataclass
class RasterDataset:
    """
    Plain, in-memory container for a raster.

    Replaces the old raw ``gdal.Dataset`` handle — no open file
    handle to manage or leak, just data. Holds everything downstream
    code (drivers, analysis functions) needs.
    """
    array: np.ndarray
    geotransform: tuple
    crs: Optional[str]      # WKT, or None if the source has no CRS
    nodata: Optional[float]
    xsize: int
    ysize: int
    band_count: int

def read_raster(path, variable=None):
    """
    Open a geospatial raster and read it fully into memory.

    Parameters
    ----------
    path : str
        Path to the geospatial dataset.
    variable : str, optional
        Multi-variable containers (NetCDF, HDF) often have no raster
        bands at the top level — only "subdatasets", one per variable.
        For those, pass the variable name to select which one to read
        (e.g. variable="precipitation"). Not needed for single-band/
        single-variable files like a plain GeoTIFF.

    Returns
    -------
    RasterDataset

    Raises
    ------
    FileNotFoundError
        If the dataset cannot be opened.
    ValueError
        If the file has no bands and no subdatasets, if it has
        subdatasets but no variable was specified, or if the given
        variable isn't among the file's subdatasets. In the latter
        two cases the error message lists what's actually available.
    """
    try:
        with rasterio.open(path) as src:
            if src.count == 0:
                if not src.subdatasets:
                    raise ValueError(f"No raster bands or subdatasets found in: {path}")

                available = [sd.split(":")[-1] for sd in src.subdatasets]

                if variable is None:
                    raise ValueError(
                        f"{path} has multiple variables and no top-level raster "
                        f"band — pass variable=<name>. Available: {available}"
                    )

                match = next(
                    (sd for sd in src.subdatasets if sd.split(":")[-1] == variable),
                    None,
                )
                if match is None:
                    raise ValueError(
                        f"Variable '{variable}' not found in {path}. "
                        f"Available: {available}"
                    )

                # Reopen directly on the subdataset path (e.g.
                # NETCDF:"file.nc":precipitation) — this one has its
                # own real bands, geotransform, and CRS.
                return read_raster(match)

            array = src.read(1) if src.count == 1 else src.read()
            return RasterDataset(
                array=array,
                geotransform=src.transform.to_gdal(),
                crs=src.crs.to_wkt() if src.crs else None,
                nodata=src.nodata,
                xsize=src.width,
                ysize=src.height,
                band_count=src.count,
            )
    except RasterioIOError as exc:
        raise FileNotFoundError(f"Unable to open dataset: {path}") from exc

def read_raster_array(path, band=None, variable=None):
    """
    Read a raster's pixel data plus the metadata needed to write it back out.

    Parameters
    ----------
    path : str
        Path to the raster file.
    band : int, optional
        1-indexed band number to read. If None, reads all bands
        (returns a 3D array when band_count > 1, 2D when band_count == 1).
    variable : str, optional
        See read_raster() — required for multi-variable NetCDF/HDF
        files with no top-level raster band.

    Returns
    -------
    dict
        {
            "array": np.ndarray,
            "geotransform": tuple,
            "crs": str | None,   # WKT
            "nodata": float | None,
            "xsize": int,
            "ysize": int,
            "band_count": int,
        }
    """
    ds = read_raster(path, variable=variable)

    if band is not None and ds.band_count > 1:
        array = ds.array[band - 1]
    else:
        array = ds.array

    return {
        "array": array,
        "geotransform": ds.geotransform,
        "crs": ds.crs,
        "nodata": ds.nodata,
        "xsize": ds.xsize,
        "ysize": ds.ysize,
        "band_count": ds.band_count,
    }

def write_raster(dataset, path, xsize, ysize, band_count=1, dtype="float32",
                  driver="GTiff", crs=None, geotransform=None, nodata=None):
    """
    Write array data to a raster file.

    Parameters
    ----------
    dataset : np.ndarray
        2D array (single band) or 3D array (band, rows, cols) of pixel values.
    path : str
        Output file path.
    xsize, ysize : int
        Raster width and height in pixels.
    band_count : int, optional
        Number of bands. Defaults to 1.
    dtype : str, optional
        Output pixel data type (numpy/rasterio dtype string, e.g.
        "float32", "uint8"). Defaults to "float32".
    driver : str, optional
        GDAL driver short name. Defaults to "GTiff".
    crs : str, optional
        Coordinate reference system as a WKT string or EPSG code
        (e.g. "EPSG:4326"). rasterio accepts both directly.
    geotransform : tuple, optional
        GDAL-style geotransform tuple
        (origin_x, pixel_width, 0, origin_y, 0, pixel_height).
    nodata : float, optional
        NoData value to set on each band.

    Returns
    -------
    str
        Path to the written file.
    """
    transform = Affine.from_gdal(*geotransform) if geotransform is not None else None

    with rasterio.open(
        path, "w",
        driver=driver,
        height=ysize,
        width=xsize,
        count=band_count,
        dtype=dtype,
        crs=crs,
        transform=transform,
        nodata=nodata,
    ) as dst:
        if band_count == 1:
            dst.write(dataset, 1)
        else:
            for i in range(1, band_count + 1):
                dst.write(dataset[i - 1], i)

    return path
