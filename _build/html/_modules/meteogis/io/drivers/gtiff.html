@register
class GTiffDriver(Driver):
    """
    Raster driver backed by rasterio, wrapping io.raster's read/write.

    read() returns a RasterDataset (array, geotransform, crs, nodata,
    xsize, ysize, band_count) — same fields as calling
    raster.read_raster() yourself, just no live file handle to manage.
    For a plain dict with the same fields, use
    raster.read_raster_array() directly.

    write() accepts either a RasterDataset (as read() returns) or a
    plain numpy array, so both styles round-trip.
    """

    name = "GTiff"
    extensions = [".tif", ".tiff"]

    def write(self, dataset, path, **overrides):
        """
        Parameters
        ----------
        dataset : RasterDataset | np.ndarray
            A RasterDataset (e.g. from read()) — metadata is pulled from
            it automatically. Or a plain array — pass xsize/ysize/crs/etc.
            via **overrides in that case, same as raster.write_raster().
        path : str
        **overrides
            Overrides any field pulled from a RasterDataset, or supplies
            the required fields when dataset is a plain array.
        """
        if isinstance(dataset, RasterDataset):
            params = {
                "xsize": dataset.xsize,
                "ysize": dataset.ysize,
                "band_count": dataset.band_count,
                "crs": dataset.crs,
                "geotransform": dataset.geotransform,
                "nodata": dataset.nodata,
            }
            array = dataset.array
        else:
            params = {}
            array = dataset

        params.update(overrides)
        return raster.write_raster(array, path, **params)
