@register
class CSVDriver(Driver):
    """
    Reads/writes point-geometry vector data as CSV, with coordinates
    stored in ``latitude``/``longitude`` columns and all other columns
    treated as feature properties.

    Only supports Point geometries — CSV has no native way to represent
    lines or polygons. For those, use GeoJSONDriver or ShapefileDriver.
    """

    name = "CSV"
    extensions = [".csv"]

    def write(self, dataset, path, lon_field="longitude", lat_field="latitude", geometry_mode="representative_point"):
        rows = []
        fieldnames = [lon_field, lat_field]

        for geometry, properties in dataset:
            
            if geometry.geom_type != "Point":
                
                if geometry_mode == "centroid":
                    geometry = geometry.centroid

                elif geometry_mode == "representative_point":
                    geometry = geometry.representative_point()

                else:
                    raise ValueError(
                        f"CSVDriver only supports Point geometries, got {geometry.geom_type}"
                    )

            row = {lon_field: geometry.x, lat_field: geometry.y}
            row.update(properties or {})
            rows.append(row)

            for key in row:
                if key not in fieldnames:
                    fieldnames.append(key)

        with open(path, "w", newline="", encoding="utf-8") as f:
            writer = csv_module.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(rows)

    def read(self, path, lon_field="longitude", lat_field="latitude"):
        dataset = []

        with open(path, "r", newline="", encoding="utf-8") as f:
            reader = csv_module.DictReader(f)
            for row in reader:
                lon = float(row.pop(lon_field))
                lat = float(row.pop(lat_field))
                geometry = Point(lon, lat)
                dataset.append((geometry, row))

        return dataset
