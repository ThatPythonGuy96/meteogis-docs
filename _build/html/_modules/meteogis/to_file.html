def to_file(dataset, file_path, **kwargs):
    """
    Write data to a file (raster or vector), auto-dispatching on file extension.

    Tries the driver registry first (covers every registered vector
    format, plus .tif/.tiff via GTiffDriver). Falls back to GDAL's
    generic raster writer for formats with no registered driver.

    Parameters
    ----------
    dataset : Vector: list[tuple[shapely.Geometry, dict]]
               Raster: gdal.Dataset | np.ndarray
        Data to write. Matches whatever the corresponding read_file()
        call for this format would have returned.
    file_path : str
        Path to write to.
    **kwargs
        Forwarded to the matching driver's write() (or write_raster()
        on fallback) — e.g. crs=<WKT or "EPSG:xxxx"> for vector formats,
        geometry_mode for CSV.
    """
    try:
        driver = get_driver(file_path)
        return driver.write(dataset, file_path, **kwargs)
    except ValueError:
        return write_raster(dataset, file_path, **kwargs)
